
#include <stdio.h>
#define MAX 10
#define INF 999

int getMinimum(int d[], int visited[], int n) {
    int min = INF, u = -1;

    for (int i = 0; i < n; i++) {
        if (!visited[i] && d[i] < min) {
            min = d[i];
            u = i;
        }
    }

    return u;
}

void linkState(int n, int cost[MAX][MAX], int source) {
    int d[MAX], visited[MAX];
    int u, v;

    for (int i = 0; i < n; i++) {
        d[i] = INF;
        visited[i] = 0;
    }

    d[source] = 0;

    for (int count = 0; count < n; count++) {
        u = getMinimum(d, visited, n);

        if (u == -1)
            break;

        visited[u] = 1;

        for (v = 0; v < n; v++) {
            if (!visited[v] && cost[u][v] != INF &&
                d[u] + cost[u][v] < d[v]) {

                d[v] = d[u] + cost[u][v];
            }
        }
    }

    printf("\nLINK STATE ROUTING TABLE\n");
    printf("Destination\tCost\n");

    for (int i = 0; i < n; i++) {
        printf("%c\t\t", 'A' + i);

        if (d[i] == INF)
            printf("INF\n");
        else
            printf("%d\n", d[i]);
    }
}

int main() {
    int n, cost[MAX][MAX], source;

    printf("Enter number of routers: ");
    scanf("%d", &n);

    if (n < 1 || n > MAX) {
        printf("Invalid number of routers!\n");
        return 1;
    }

    printf("Enter cost matrix (999 for no link):\n");

    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            scanf("%d", &cost[i][j]);

    printf("Enter source router (0 to %d): ", n - 1);
    scanf("%d", &source);

    if (source < 0 || source >= n) {
        printf("Invalid source router!\n");
        return 1;
    }

    linkState(n, cost, source);

    return 0;
}


























































#include <stdio.h>
#define MAX 10
#define INF 999

int main() {
    int n, cost[MAX][MAX], dist[MAX][MAX];
    int i, j, k, changed;

    printf("Enter number of routers: ");
    scanf("%d", &n);

    printf("Enter cost matrix (999 for no link):\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++) {
            scanf("%d", &cost[i][j]);
            dist[i][j] = cost[i][j];
        }

    do {
        changed = 0;

        for (i = 0; i < n; i++)
            for (j = 0; j < n; j++)
                for (k = 0; k < n; k++) {
                    if (dist[i][k] != INF &&
                        dist[k][j] != INF &&
                        dist[i][k] + dist[k][j] < dist[i][j]) {

                        dist[i][j] = dist[i][k] + dist[k][j];
                        changed = 1;
                    }
                }
    } while (changed);

    printf("\nDISTANCE VECTOR ROUTING TABLE\n");

    for (i = 0; i < n; i++) {
        printf("\nRouter %c\n", 'A' + i);
        printf("Destination\tCost\n");

        for (j = 0; j < n; j++) {
            printf("%c\t\t", 'A' + j);

            if (dist[i][j] == INF)
                printf("INF\n");
            else
                printf("%d\n", dist[i][j]);
        }
    }

    return 0;
}
