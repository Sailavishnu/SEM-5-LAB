PROGRAM 1: ROLL NUMBER EXTRACTOR
================================================================================
%{
#include<stdio.h>
#include<string.h>
#include<ctype.h>

FILE *out;
int count = 0;
%}

%%

[0-9]{2}[A-Z]{3}[0-9]{3}  {
    fprintf(out, "%s\n", yytext);
    count++;
}

[ \t]+ ;

\n      ;

.       ;

%%

int yywrap()
{
    return 1;
}

int main()
{
    FILE *in;
    
    in = fopen("student.txt", "r");
    if(in == NULL)
    {
        printf("Cannot open input file.\n");
        return 1;
    }
    
    out = fopen("rollnumbers.txt", "w");
    if(out == NULL)
    {
        printf("Cannot open output file.\n");
        return 1;
    }
    
    yyin = in;
    yylex();
    
    printf("Roll Numbers found:\n");
    printf("Total number of students = %d\n", count);
    
    fclose(in);
    fclose(out);
    
    return 0;
}

================================================================================
PROGRAM 2: HTML HYPERLINK EXTRACTOR
================================================================================
%{
#include<stdio.h>
#include<string.h>
#include<stdlib.h>

FILE *out;
char link_url[500];
int in_anchor = 0;

void extractURL(char *str)
{
    char *start = strstr(str, "href=\"");
    if(start != NULL)
    {
        start += 6;
        int i = 0;
        while(start[i] != '"' && start[i] != '\0')
        {
            link_url[i] = start[i];
            i++;
        }
        link_url[i] = '\0';
    }
}
%}

%%

"<a"[^>]*"href="[^>]*">"  {
    in_anchor = 1;
    extractURL(yytext);
}

"</a>"  {
    in_anchor = 0;
}

[^<\n]+  {
    if(in_anchor && link_url[0] != '\0')
    {
        fprintf(out, "Link Text: %s\n", yytext);
        fprintf(out, "URL: %s\n\n", link_url);
        link_url[0] = '\0';
    }
}

.|\n  ;

%%

int yywrap()
{
    return 1;
}

int main()
{
    FILE *in;
    
    in = fopen("webpage.html", "r");
    if(in == NULL)
    {
        printf("Cannot open HTML file.\n");
        return 1;
    }
    
    out = fopen("links.txt", "w");
    if(out == NULL)
    {
        printf("Cannot open output file.\n");
        return 1;
    }
    
    yyin = in;
    yylex();
    
    printf("Hyperlinks extracted successfully.\n");
    
    fclose(in);
    fclose(out);
    
    return 0;
}

================================================================================
PROGRAM 3: TEXT MODIFIER (is→was, increment numbers)
================================================================================
%{
#include<stdio.h>
#include<string.h>
#include<stdlib.h>

FILE *out;
int is_count = 0;

void incrementNumber(char *num_str)
{
    int num = atoi(num_str);
    num += 10;
    fprintf(out, "%d", num);
}
%}

%%

"is"  {
    fprintf(out, "was");
    is_count++;
}

[0-9]+  {
    incrementNumber(yytext);
}

.  {
    fprintf(out, "%s", yytext);
}

\n  {
    fprintf(out, "\n");
}

%%

int yywrap()
{
    return 1;
}

int main()
{
    FILE *in;
    
    in = fopen("input_text.txt", "r");
    if(in == NULL)
    {
        printf("Cannot open input file.\n");
        return 1;
    }
    
    out = fopen("output_text.txt", "w");
    if(out == NULL)
    {
        printf("Cannot open output file.\n");
        return 1;
    }
    
    yyin = in;
    yylex();
    
    printf("Text processing completed.\n");
    printf("Number of 'is' replaced = %d\n", is_count);
    
    fclose(in);
    fclose(out);
    
    return 0;
}
